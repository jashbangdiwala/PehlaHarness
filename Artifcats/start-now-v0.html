Start now: harness v0

A Telegram bot on your Mac that only answers you, sends your messages to a model through one guarded function, and logs everything. About 45 minutes. Do the steps in order.

5events.py: the log

Every step is a row. This is your single source of truth.

events.py
import sqlite3, json, time

db = sqlite3.connect("ea.db")
db.execute("""CREATE TABLE IF NOT EXISTS events(
  id INTEGER PRIMARY KEY, ts REAL, kind TEXT,
  domain TEXT, payload TEXT, parent_id INTEGER)""")

def log(kind, payload, domain="core", parent_id=None):
    cur = db.execute(
        "INSERT INTO events(ts,kind,domain,payload,parent_id) VALUES(?,?,?,?,?)",
        (time.time(), kind, domain, json.dumps(payload), parent_id))
    db.commit()
    return cur.lastrowid

6egress.py: the one door to any model

Nothing else in your code is allowed to call a model. Redaction and logging happen here.

egress.py
import re, time, httpx, keyring
from events import log

# Use a model ID copied from openrouter.ai/models. This is an example, verify it exists.
MODEL = "meta-llama/llama-3.3-70b-instruct"

PATTERNS = [
    (re.compile(r"\b(?:\d[ -]?){13,19}\b"), "[CARD]"),
    (re.compile(r"\bsk-[A-Za-z0-9_-]{16,}"), "[KEY]"),
    # add your own: national ID formats, account numbers, etc.
]

def redact(text):
    for pat, tag in PATTERNS:
        text = pat.sub(tag, text)
    return text

def ask(messages, parent_id=None):
    safe = [{"role": m["role"], "content": redact(m["content"])} for m in messages]
    key = keyring.get_password("ea", "openrouter")
    t = time.time()
    r = httpx.post(
        "https://openrouter.ai/api/v1/chat/completions",
        headers={"Authorization": f"Bearer {key}"},
        json={"model": MODEL, "messages": safe,
              "provider": {"data_collection": "deny", "allow_fallbacks": False}},
        timeout=60)
    r.raise_for_status()
    data = r.json()
    text = data["choices"][0]["message"]["content"]
    log("model_call", {"model": MODEL, "secs": round(time.time() - t, 2),
                       "usage": data.get("usage")}, parent_id=parent_id)
    return text
If you get a "no endpoints" error: that model has no provider meeting the data_collection: deny setting. Pick another model. I have not verified these parameter names against current docs, so check OpenRouter's provider-routing page if it errors in another way.

7main.py: Telegram loop

main.py
import os, time, httpx, keyring
from events import log
from egress import ask

TOKEN = keyring.get_password("ea", "telegram")
API = f"https://api.telegram.org/bot{TOKEN}"
ID_FILE = "allowed_id.txt"
ALLOWED = int(open(ID_FILE).read().strip()) if os.path.exists(ID_FILE) else None

def scrub(e):
    return repr(e).replace(TOKEN, "***")

def send(chat_id, text):
    httpx.post(f"{API}/sendMessage",
               json={"chat_id": chat_id, "text": text}, timeout=30)

def handle(update):
    msg = update.get("message")
    if not msg or "text" not in msg:
        return
    uid = msg["from"]["id"]
    if ALLOWED is None:
        print(f"Your Telegram user ID is {uid}. Put it in {ID_FILE} and restart.")
        log("dropped", {"reason": "no allowlist yet"})
        return
    if uid != ALLOWED:
        log("dropped", {"reason": "not allowed"})   # silent: no reply, no content stored
        return
    mid = log("message", {"text": msg["text"]})
    reply = ask([
        {"role": "system", "content": "You are a concise personal assistant."},
        {"role": "user", "content": msg["text"]},
    ], parent_id=mid)
    log("reply", {"text": reply}, parent_id=mid)
    send(msg["chat"]["id"], reply)

def main():
    offset = None
    print("polling... (Ctrl-C to stop)")
    while True:
        params = {"timeout": 30}
        if offset:
            params["offset"] = offset
        try:
            r = httpx.get(f"{API}/getUpdates", params=params, timeout=40)
            for u in r.json().get("result", []):
                offset = u["update_id"] + 1
                try:
                    handle(u)
                except Exception as e:
                    log("error", {"err": scrub(e)})
        except httpx.HTTPError as e:
            log("error", {"err": scrub(e)})
            time.sleep(5)

if __name__ == "__main__":
    main()

from egress import redact

def test_card_is_redacted():
    assert "4111 1111 1111 1111" not in redact("pay 4111 1111 1111 1111 now")

def test_key_is_redacted():
    assert "sk-abcdefghijklmnop1234" not in redact("my key sk-abcdefghijklmnop1234")
